{# One data cell of a rendered frame. Its own file so a template can `{% from %}` it: importing a macro executes the template it sits in, and _stage_diff.html's body needs a `diff` to render. #} {% macro data_cell(text, klass, was, full, links) -%} {%- if text.cites is defined %} {%- if was %}{{ was | truncate(120) }}{% endif %} {{- span_cites(text, links) }} {%- else %} {%- set klasses = (('cell-clip ' if full else '') ~ klass) | trim -%} {#- The title is the hover: it is what a reader gets instead of expanding a cell, so it is written wherever the cell below had to cut the value. -#} {%- set cut = full or text | length > 240 -%} {%- if was %}{{ was | truncate(120) }}{% endif %} {%- if full %}{{ text | truncate(4000, True, '… (truncated — download the CSV for the full value)') }} {%- else %}{{ text | truncate(240) }}{% endif %} {%- endif %} {%- endmacro %} {# A span cell (app/web/span_cells.py): each quote, cut short with the whole in its title, then where it sits, linked to that page of its file. #} {% macro span_cites(cell, links) -%} {%- for cite in cell.cites %}{{ cite.span.quote | truncate(120) }} {%- set href = links.source_page(cite.span) %} {% if href %}{{ cite.label }}{% else %}{{ cite.label }}{% endif %} {%- endfor %} {%- endmacro %}
